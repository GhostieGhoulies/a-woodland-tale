    function selectCharacter(charName) {
      selectedChar = charName;

    function returnToSelect() {
      gameState = 'SELECT';
      document.getElementById('gameOverScreen').style.display = 'none';
      document.getElementById('hud').style.display = 'none';
      document.getElementById('selectScreen').style.display = 'flex';
    }

    function prepareRun() {
      gameState = 'READY';
      score = 0;
      gameSpeed = 5.5;
      obstacles = [];
      nextSpawnTimer = 70;
      player.y = groundY - 22;
      player.vy = 0;
      player.isGrounded = true;
      player.isDucking = false;

      document.getElementById('gameOverScreen').style.display = 'none';
      document.getElementById('hud').style.display = 'block';
      document.getElementById('scoreDisplay').textContent = '0';
      document.getElementById('instructions').style.display = 'block';
    }

    function startRun() {
      if (gameState !== 'READY') return;
      gameState = 'PLAYING';
      document.getElementById('instructions').style.display = 'none';
    }

    // Touch / Swipe Input Handling
    let touchStartY = 0;
    let touchStartX = 0;

    window.addEventListener('touchstart', (e) => {
      if (gameState === 'READY') {
        startRun();
        return;
      }
      if (gameState === 'PLAYING') {
        touchStartY = e.touches[0].clientY;
        touchStartX = e.touches[0].clientX;
      }
    }, { passive: true });

    window.addEventListener('touchmove', (e) => {
      if (gameState !== 'PLAYING') return;
      const dy = e.touches[0].clientY - touchStartY;

      if (dy < -28) {
        // Swipe Up -> Jump
        jump();
        touchStartY = e.touches[0].clientY;
      } else if (dy > 28) {
        // Swipe Down -> Duck
        duck();
        touchStartY = e.touches[0].clientY;
      }
    }, { passive: true });

    // Optional keyboard support if tested on desktop
    window.addEventListener('keydown', (e) => {
      if (gameState === 'READY' && (e.code === 'Space' || e.code === 'ArrowUp')) {
        startRun();
      } else if (gameState === 'PLAYING') {
        if (e.code === 'ArrowUp' || e.code === 'Space') jump();
        if (e.code === 'ArrowDown') duck();
      }
    });

    function jump() {
      if (player.isGrounded) {
        player.vy = -13.2;
        player.isGrounded = false;
        player.isDucking = false;
      }
    }

    function duck() {
      player.isDucking = true;
      player.duckTimer = 42; // Duck duration in frames
      if (!player.isGrounded) {
        // Fast-drop if swiping down mid-air
        player.vy = 10;
      }
    }

    function spawnObstacle() {
      // 6 distinct obstacles:
      // Jump over: 'log', 'briar', 'puddle', 'fence'
      // Duck under: 'bird', 'branch'
      const types = ['log', 'briar', 'puddle', 'fence', 'bird', 'branch'];
      const type = types[Math.floor(Math.random() * types.length)];

      let obs = {
        type: type,
        x: width + 80,
        passed: false
      };

      if (type === 'log') {
        obs.w = 42;
        obs.h = 30;
        obs.y = groundY - obs.h;
      } else if (type === 'briar') {
        obs.w = 50;
        obs.h = 28;
        obs.y = groundY - obs.h;
      } else if (type === 'puddle') {
        obs.w = 64;
        obs.h = 12;
        obs.y = groundY - 6;
      } else if (type === 'fence') {
        obs.w = 34;
        obs.h = 42;
        obs.y = groundY - obs.h;
      } else if (type === 'bird') {
        // Low flying bird positioned at head height (duck to avoid)
        obs.w = 44;
        obs.h = 24;
        obs.y = groundY - 52;
      } else if (type === 'branch') {
        // Background tree with low overhanging branch (duck to avoid)
        obs.w = 56;
        obs.h = 30;
        obs.y = groundY - 56;
      }

      obstacles.push(obs);
    }

    function drawStaticBackground() {
      // Warm cottagecore golden-hour forest sky/mist gradient
      const skyGrad = ctx.createLinearGradient(0, 0, 0, groundY);
      skyGrad.addColorStop(0, '#4f6d4a');
      skyGrad.addColorStop(0.55, '#8aa373');
      skyGrad.addColorStop(1, '#d8cfa8');
      ctx.fillStyle = skyGrad;
      ctx.fillRect(0, 0, width, groundY);

      // Soft sunbeams filtering through canopy
      ctx.fillStyle = 'rgba(255, 244, 200, 0.12)';
      ctx.beginPath();
      ctx.moveTo(width * 0.2, 0);
      ctx.lineTo(width * 0.45, 0);
      ctx.lineTo(width * 0.65, groundY);
      ctx.lineTo(width * 0.1, groundY);
      ctx.closePath();
      ctx.fill();

      // Distant woodland hills
      ctx.fillStyle = '#5c7852';
      ctx.beginPath();
      ctx.moveTo(0, groundY);
      for (let x = 0; x <= width; x += 60) {
        ctx.lineTo(x, groundY - 65 - Math.sin(x * 0.01) * 25);
      }
      ctx.lineTo(width, groundY);
      ctx.fill();

      // Stationary forest trees & foliage
      bgDetails.forEach(item => {
        if (item.type === 'bgTree') {
          // Trunk
          ctx.fillStyle = '#4a3525';
          ctx.fillRect(item.x - item.trunkW / 2, groundY - 190, item.trunkW, 190);
          // Canopy
          ctx.fillStyle = item.shade;
          ctx.beginPath();
          ctx.arc(item.x, groundY - 195, item.canopyR, 0, Math.PI * 2);
          ctx.arc(item.x - 30, groundY - 175, item.canopyR * 0.75, 0, Math.PI * 2);
          ctx.arc(item.x + 30, groundY - 175, item.canopyR * 0.75, 0, Math.PI * 2);
          ctx.fill();
        } else if (item.type === 'mushroom') {
          ctx.save();
          ctx.translate(item.x, groundY);
          ctx.scale(item.scale, item.scale);
          ctx.fillStyle = '#f5ebe0';
          ctx.fillRect(-4, -14, 8, 14);
          ctx.fillStyle = '#c14953';
          ctx.beginPath();
          ctx.arc(0, -14, 12, Math.PI, 0);
          ctx.fill();
          ctx.fillStyle = '#fff';
          ctx.beginPath();
          ctx.arc(-4, -18, 2, 0, Math.PI * 2);
          c.arc(4, -17, 1.8, 0, Math.PI * 2);
          ctx.fill();
          ctx.restore();
        } else if (item.type === 'fern') {
          ctx.fillStyle = '#3d6336';
          ctx.beginPath();
          ctx.ellipse(item.x - 7, groundY - 6, 10, 4, -0.4, 0, Math.PI * 2);
          ctx.ellipse(item.x + 7, groundY - 6, 10, 4, 0.4, 0, Math.PI * 2);
          ctx.fill();
        } else if (item.type === 'flowers') {
          ctx.fillStyle = '#4f772d';
          ctx.fillRect(item.x - 1, groundY - 10, 2, 10);
          ctx.fillStyle = item.color;
          ctx.beginPath();
          ctx.arc(item.x, groundY - 11, 4, 0, Math.PI * 2);
          ctx.fill();
        }
      });
    }

    function drawForestFloor() {
      // Base rich soil track
      ctx.fillStyle = '#4b3621';
      ctx.fillRect(0, groundY, width, height - groundY);

      // Mossy grass top border
      ctx.fillStyle = '#607c3c';
      ctx.fillRect(0, groundY, width, 12);

      // Moving floor texture (pebbles, fallen leaves, roots moving left)
      const spacing = 48;
      const startX = -(floorOffset % spacing);
      for (let x = startX; x < width + spacing; x += spacing) {
        // Grass tuft
        ctx.fillStyle = '#739347';
        ctx.beginPath();
        ctx.moveTo(x, groundY);
        ctx.lineTo(x + 6, groundY - 6);
        ctx.lineTo(x + 12, groundY);
        ctx.fill();

        // Soil pebbles & twigs
        ctx.fillStyle = '#362515';
        ctx.beginPath();
        ctx.ellipse(x + 18, groundY + 26, 7, 3.5, 0, 0, Math.PI * 2);
        ctx.fill();

        ctx.fillStyle = '#6b4f35';
        ctx.fillRect(x + 32, groundY + 45, 14, 3);
      }
    }

    function drawObstacles() {
      obstacles.forEach(obs => {
        if (obs.type === 'log') {
          // Fallen Log (Hop over)
          ctx.fillStyle = '#5c3a21';
          ctx.beginPath();
          ctx.roundRect(obs.x, obs.y, obs.w, obs.h, 8);
          ctx.fill();
          // Bark lines & end ring
          ctx.strokeStyle = '#3b2312';
          ctx.lineWidth = 2;
          ctx.strokeRect(obs.x + 4, obs.y + 6, obs.w - 16, 4);
          ctx.fillStyle = '#c89f72';
          ctx.beginPath();
          ctx.ellipse(obs.x + obs.w - 7, obs.y + obs.h / 2, 6, obs.h / 2 - 2, 0, 0, Math.PI * 2);
          ctx.fill();

        } else if (obs.type === 'briar') {
          // Thorny Briar Patch (Hop over)
          ctx.fillStyle = '#2d4726';
          ctx.beginPath();
          ctx.arc(obs.x + 12, obs.y + 18, 13, 0, Math.PI * 2);
          ctx.arc(obs.x + 26, obs.y + 14, 15, 0, Math.PI * 2);
          ctx.arc(obs.x + 39, obs.y + 19, 12, 0, Math.PI * 2);
          ctx.fill();
          // Thorns & berries
          ctx.fillStyle = '#9e2a2b';
          ctx.beginPath();
          ctx.arc(obs.x + 20, obs.y + 12, 3, 0, Math.PI * 2);
          ctx.arc(obs.x + 33, obs.y + 16, 3, 0, Math.PI * 2);
          ctx.fill();

        } else if (obs.type === 'puddle') {
          // Water Puddle (Hop over)
          ctx.fillStyle = '#5fa8d3';
          ctx.beginPath();
          ctx.ellipse(obs.x + obs.w / 2, groundY + 5, obs.w / 2, 8, 0, 0, Math.PI * 2);
          ctx.fill();
          ctx.strokeStyle = '#bee9e8';
          ctx.lineWidth = 2;
          ctx.beginPath();
          ctx.ellipse(obs.x + obs.w / 2, groundY + 5, obs.w / 3, 4, 0, 0, Math.PI * 2);
          ctx.stroke();

        } else if (obs.type === 'fence') {
          // Cottagecore Wooden Picket Fence (Hop over)
          ctx.fillStyle = '#e6ccb2';
          // Horizontal rails
          ctx.fillRect(obs.x, obs.y + 12, obs.w, 5);
          ctx.fillRect(obs.x, obs.y + 28, obs.w, 5);
          // Vertical pickets
          for (let px = obs.x + 2; px < obs.x + obs.w - 4; px += 11) {
            ctx.beginPath();
            ctx.moveTo(px, obs.y + obs.h);
            ctx.lineTo(px, obs.y + 6);
            ctx.lineTo(px + 4, obs.y);
            ctx.lineTo(px + 8, obs.y + 6);
            ctx.lineTo(px + 8, obs.y + obs.h);
            ctx.closePath();
            ctx.fill();
          }

        } else if (obs.type === 'bird') {
          // Low-Flying Woodland Bird (Duck under)
          const wingFlap = Math.sin(frameCount * 0.25) * 8;
          ctx.fillStyle = '#bc4749'; // Robin red-brown
          ctx.beginPath();
          ctx.ellipse(obs.x + 22, obs.y + 14, 16, 9, 0, 0, Math.PI * 2);
          ctx.fill();
          // Beak facing left
          ctx.fillStyle = '#ffb703';
          ctx.beginPath();
          ctx.moveTo(obs.x + 6, obs.y + 12);
          ctx.lineTo(obs.x - 2, obs.y + 15);
          ctx.lineTo(obs.x + 6, obs.y + 17);
          ctx.closePath();
          ctx.fill();
          // Wing
          ctx.strokeStyle = '#6a2c2e';
          ctx.lineWidth = 3.5;
          ctx.beginPath();
          ctx.moveTo(obs.x + 20, obs.y + 12);
          ctx.lineTo(obs.x + 28, obs.y + 12 - wingFlap);
          ctx.stroke();

        } else if (obs.type === 'branch') {
          // Tree behind track with low overhanging branch (Duck under)
          // Tree trunk rooted behind the track
          ctx.fillStyle = '#3f2a1d';
          ctx.fillRect(obs.x + 36, obs.y - 110, 22, 110 + (groundY - obs.y));
          // Low overhanging branch extending left across player head height
          ctx.fillStyle = '#4f3524';
          ctx.beginPath();
          ctx.roundRect(obs.x, obs.y + 6, 46, 12, 5);
          ctx.fill();
          // Leaf cluster hanging from branch
          ctx.fillStyle = '#588157';
          ctx.beginPath();
          ctx.arc(obs.x + 14, obs.y + 12, 13, 0, Math.PI * 2);
          ctx.arc(obs.x + 28, obs.y + 10, 14, 0, Math.PI * 2);
          ctx.fill();
        }
      });
    }

    function update() {
      frameCount++;

      if (gameState !== 'PLAYING') return;

      // Move floor left & increase speed gradually
      floorOffset += gameSpeed;
      gameSpeed += 0.0008;

      // Distance-based score
      score += gameSpeed * 0.03;
      document.getElementById('scoreDisplay').textContent = Math.floor(score);

      // Player gravity & jump
      if (!player.isGrounded) {
        player.vy += 0.65;
        player.y += player.vy;

        if (player.y >= groundY - 22) {
          player.y = groundY - 22;
          player.vy = 0;
          player.isGrounded = true;
        }
      }

      // Duck timer
      if (player.isDucking) {
        player.duckTimer--;
        if (player.duckTimer <= 0) {
          player.isDucking = false;
        }
      }

      // Nicely spaced obstacle spawning (adjusts with speed so spacing stays fair)
      nextSpawnTimer--;
      if (nextSpawnTimer <= 0) {
        spawnObstacle();
        const minGap = Math.max(68, 125 - gameSpeed * 4);
        nextSpawnTimer = minGap + Math.random() * 45;
      }

      // Move obstacles & check collisions
      const pHeight = player.isDucking ? player.duckHeight : player.normalHeight;
      const pBox = {
        left: player.x - 14,
        right: player.x + 14,
        top: (player.y + 22) - pHeight,
        bottom: player.y + 20
      };

      for (let i = obstacles.length - 1; i >= 0; i--) {
        let obs = obstacles[i];
        obs.x -= gameSpeed;

        // Forgiving hitbox padding
        const oBox = {
          left: obs.x + 5,
          right: obs.x + obs.w - 5,
          top: obs.y + 4,
          bottom: obs.y + obs.h - 2
        };

        if (
          pBox.right > oBox.left &&
          pBox.left < oBox.right &&
          pBox.bottom > oBox.top &&
          pBox.top < oBox.bottom
        ) {
          endRun();
          return;
        }

        if (obs.x + obs.w < -80) {
          obstacles.splice(i, 1);
        }
      }
    }

    function endRun() {
      gameState = 'GAMEOVER';
      const finalDist = Math.floor(score);
      if (finalDist > bestScore) bestScore = finalDist;

      document.getElementById('finalScore').textContent = finalDist;
      document.getElementById('bestScoreText').textContent = `Best: ${bestScore}m`;
      document.getElementById('gameOverScreen').style.display = 'flex';
    }

    function loop() {
      ctx.clearRect(0, 0, width, height);

      drawStaticBackground();
      drawForestFloor();

      if (gameState !== 'SELECT') {
        drawObstacles();
        const animTick = gameState === 'PLAYING' ? frameCount : 0;
        drawCharacter(ctx, selectedChar, player.x, player.y, player.isDucking, animTick, player.isGrounded);
      }

      update();
      requestAnimationFrame(loop);
    }

    loop();
  </script>
